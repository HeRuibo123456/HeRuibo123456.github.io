P1045 [NOIP 2003 普及组] 麦森数

题目大意

要求出2P-1的位数和后500位的结果(P < 3100000)

题目思路

这道题一看就需要用到高精度乘法，因为2的次幂的个位不可能是0，所以直接在最后一位减一即可

交上去，超时，喜提50pts

我们再仔细地想一想，P < 3.1 * 106，贸然高精度乘法肯定会超时

所以，就可以用到快速幂来快速求解

实现细节

因为要用到快速幂+高精度，所以可以把高精度封装在一个结构体中，这样直接写快速幂板子即可，非常省事

为了节省运行时的内存，我们可以把数组大小设成500，每次只计算后500位的数字，提升了效率

因为我们只算了500位，所以我们要计算它的位数。我去请教了DeepSeek，他说， 计算某个数x的位数要用floor(log10x) + 1, 把2n代入即可，得这个数的位数= floor(n * log102) + 1

代码

#include<iostream>
#include<cmath>
#define MAX 502
using namespace std;

struct BigInt {
	int num[MAX] = {0}, len = 0;
	BigInt(int n = 0) {
		len = 0;
		while(n) {
			num[++len] = n % 10;
			n /= 10;
		}
	}
	int & operator [] (int index) {
		return num[index];
	}
};

BigInt operator * (BigInt a, BigInt b) {
	BigInt c;
	c.len = min(500, a.len + b.len);
	for(int i = 1; i <= 500; i++) {
		for(int j = 1; j <= 500 && i + j - 1 <= 500; j++) {
			c[i + j - 1] += a[i] * b[j];
			c[i + j] += c[i + j - 1] / 10;
			c[i + j - 1] %= 10;
		}
	}
	for(int i = 1; i <= c.len; i++) {
		c[i + 1] += c[i] / 10;
		c[i] %= 10;
	}
	while(c[c.len] == 0 && c.len > 1)
		c.len--;
	return c;
}

int p, digit;
BigInt result(1), b(2);

int main () {
	cin >> p;
	digit = floor(log10(2) * p) + 1;
	while(p) {
		if(p & 1) {
			result = result * b;
		}
		b = b * b;
		p >>= 1;
	}
	result[1] -= 1;
	cout << digit << endl;
	for(int i = 500; i >= 1; i--) {
		cout << result[i];
		if((i - 1) % 50 == 0)
			cout << endl;
	}
	return 0;
}
