Dijkstra - 图的最短路算法

#include<iostream>
#include<algorithm>
#include<queue>
#include<climits>
#include<cstring>
#include<utility>
#define MAX 100005
#define INF1 0x3F
#define INF2 0x3F3F3F3F
#define pii pair<int, int>
using namespace std;

vector<pii> v[MAX];
priority_queue<pii, vector<pii>, greater<pii> > pq;
int dist[MAX], n, m, s, u, v_, w;
bool done[MAX];

void dijkstra() {
	memset(dist, INF1, sizeof(dist));
	dist[s] = 0;
	pq.push({0, s});
	while(!pq.empty()) {
		pii node = pq.top();
		pq.pop();
		if(done[node.second])
			continue;
		done[node.second] = true;
		for(pii x : v[node.second]) {
			if(dist[node.second] + x.first < dist[x.second]) {
				dist[x.second] = dist[node.second] + x.first;
				pq.push({dist[x.second], x.second});
			}
		}
	}
}

int main() {
	cin >> n >> m >> s;
	for(int i = 1; i <= m; i++) {
		cin >> u >> v_ >> w;
		v[u].push_back({w, v_});
	}
	dijkstra();
	for(int i = 1; i <= n; i++) {
		if(dist[i] == INF2) {
			cout << INT_MAX << ' ';
		} else {
			cout << dist[i] << ' ';
		}
	}
	return 0;
}
